    .局部变量 hHeap, 整数型
    .局部变量 pe, 整数型

    hHeap ＝ _Get (hHashTable, #ht_hHeap)
    oldCapacity ＝ _Get (hHashTable, #ht_tableLenth) ' ht.tableLenth
    oldTable ＝ _Get (hHashTable, #ht_table) ' ht.table
    
    newCapacity ＝ oldCapacity × 2 ＋ 1
    newTable ＝ HeapAlloc (hHeap, #HEAP_ZERO_MEMORY, newCapacity × 4)
    
    _Set (hHashTable, #ht_threshold, newCapacity × _GetFloat (hHashTable, #ht_loadFactor)) ' ht.threshold ＝ newCapacity × ht.loadFactor
    _Set (hHashTable, #ht_table, newTable) ' ht.table ＝ newTable
    _Set (hHashTable, #ht_tableLenth, newCapacity) ' ht.tableLenth ＝ newCapacity
    
    .判断循环首 (i ＜ oldCapacity)
        pe ＝ _GetTableItem (oldTable, i)
        .判断循环首 (pe ≠ 0)
            oldNext ＝ _Get (pe, #jv_next) ' 获取原来的next
            index ＝ 位与 (_Get (pe, #jv_hashCode), 2147483647) ％ newCapacity ' 计算新的索引
            ' 插入到新的链表中
            _Set (pe, #jv_next, _GetTableItem (newTable, index)) ' e.next = newtable[index]
            _SetTableItem (newTable, index, pe) ' newtable[index] = e
            pe ＝ oldNext
        .判断循环尾 ()
        i ＝ i ＋ 1
    .判断循环尾 ()
    HeapFree (hHeap, 0, oldTable)
